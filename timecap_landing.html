def capture_tick(self, _):
  path = FileManager.new_capture_path()
  result = subprocess.run(
    ["screencapture", "-x",
     "-t", "jpg", path],
    timeout=15
  )

manual
Block 3
Archive Move
Moves used jpgs into archive/YYYYMM/. Tidy by month.
manual
Block 5
Log
Failures, idle skips, timeouts — recorded daily in logs/.
